extract_all_model_coefficients_ordered <- function(mdls) {
  all_names <- c(names(as.list(mdls[1]$coefficients)))
  for (mdl in mdls[1:length(mdls)]) {
    all_names <- unique( c( all_names, names(as.list(mdl$coefficients)) ) )
  }
  
  new_list <- c()
  for (name in all_names) {
    if (!(name %in% new_list)) {
      all_relevant <- grep(name, all_names, fixed = T)
      new_list <- c(new_list, all_names[all_relevant])
    }
  }
  
  new_list <- unique(new_list)
  return( list(all_names, new_list) )
}

naming_correction_2 <- function(vars, exceptions) {
  new_vars <- c()
  for (var in vars) {
    if ( grepl(var, paste(changes, collapse = ' ')) ) {
      for (change in changes) {
        if (str_split(change, ' -> ')[[1]][1] == var) {
          new_vars <- c(new_vars, str_split(change, ' -> ')[[1]][2])
        }
      }
    }
    
    else if (str_detect(var, year_regex)) {
      new_vars <- c(new_vars, str_split(var, 'as\\.factor\\(year\\)')[[1]][2])
    }
    
    else if (str_detect(var, nonlin_regex) | str_detect(var, inter_regex)) {
      inner_portion <- str_split(var, c('\\(|\\)'))[[1]][2]
      inner_portion <- convert_var_to_words(inner_portion)
      inner_portion <- gsub('\\^', "$^$", inner_portion)
      inner_portion <- gsub('\\*', "$\\\\times$", inner_portion)
      new_vars <- c(new_vars, inner_portion)
    }
    
    else {
      new_vars <- c(new_vars, convert_var_to_words(var))
    }
  }
  return(new_vars)
}

naming_correction <- function(vars, exceptions) {

  immune_list <- list()
  for (change in exceptions) {
    before <- str_split(change, ' -> ')[[1]][1]
    if (before %in% as.list(vars)) {
      immune_list[[length(immune_list) + 1]] = match(c(before), vars)
      vars[match(c(before), vars)] <- str_split(change, ' -> ')[[1]][2]
    }
  }
  
  for (i in seq(1, length(vars))) {
    if (i %in% immune_list) next
    vars[i] <- str_to_title(gsub('_', ' ', vars[i]))
    vars[i] <- gsub('I\\(', '', vars[i])
    vars[i] <- gsub('\\)', '', vars[i])
    vars[i] <- gsub('As.factor\\(Year', '', vars[i])
  }
  return(as.character(vars))
}

# Returns a Regression Formula from a regressand and a list of regressors
regression_formula <- function(dep_var, indep_vars) {
  return( as.formula(paste(dep_var, paste(unlist(c(indep_vars), use.names = F), collapse = ' + '), sep = ' ~ ')) )
}
   
# Update for Formulas so that the order is correct
update_formula_cleanly <- function(og_formula, additions) {
  split_form_parts <- as.character((og_formula))
  indep_vars <- as.list(strsplit(split_form_parts[3], ' + ', fixed = T))
  
  all_terms <- c()
  for (var in indep_vars[[1]]) {
    findings <- as.list((as.data.frame(additions) %>% filter(grepl(var, additions)))$additions)
    all_terms <- c(all_terms, c(var), as.vector(findings))
  }
  final_formula_str <- paste( split_form_parts[2], split_form_parts[1], paste(unlist(all_terms, use.names = F), collapse = ' + ') )
  return(as.formula(final_formula_str))
}

iterate_all_indiv <- function(dep_var, indep_vars, control_vars) {
  list_of_formulas <- list(length(vec_power))
  
  counter <- 1
  for (var in indep_vars) {
    form <- regression_formula(dep_var, c(var, control_vars))
    list_of_formulas[[counter]] = form; counter = counter + 1
  }
  return (list_of_formulas)
}

# To Return all Possible Model Specs
iterate_all_model_specs <- function(og_formula, set_of_specs) {
  vec_power <- powerSet(set_of_specs); list_of_formulas <- list(length(vec_power))
  
  counter <- 1
  for (set in vec_power) {
    form <- update_formula_cleanly(og_formula, set)
    list_of_formulas[[counter]] = form; counter = counter + 1
  }
  return (list_of_formulas)
}

correlation_matrix <- function(df, 
                               type = "pearson",
                               digits = 3, 
                               decimal.mark = ".",
                               use = "all", 
                               show_significance = TRUE, 
                               replace_diagonal = FALSE, 
                               replacement = ""){
  
  # check arguments
  stopifnot({
    is.numeric(digits)
    digits >= 0
    use %in% c("all", "upper", "lower")
    is.logical(replace_diagonal)
    is.logical(show_significance)
    is.character(replacement)
  })
  # we need the Hmisc package for this
  require(Hmisc)
  
  # retain only numeric and boolean columns
  isNumericOrBoolean = vapply(df, function(x) is.numeric(x) | is.logical(x), logical(1))
  if (sum(!isNumericOrBoolean) > 0) {
    cat('Dropping non-numeric/-boolean column(s):', paste(names(isNumericOrBoolean)[!isNumericOrBoolean], collapse = ', '), '\n\n')
  }
  df = df[isNumericOrBoolean]
  
  # transform input data frame to matrix
  x <- as.matrix(df)
  
  # run correlation analysis using Hmisc package
  correlation_matrix <- Hmisc::rcorr(x, type = )
  R <- correlation_matrix$r # Matrix of correlation coeficients
  p <- correlation_matrix$P # Matrix of p-value 
  
  # transform correlations to specific character format
  Rformatted = formatC(R, format = 'f', digits = digits, decimal.mark = decimal.mark)
  
  # if there are any negative numbers, we want to put a space before the positives to align all
  if (sum(R < 0) > 0) {
    Rformatted = ifelse(R > 0, paste0(' ', Rformatted), Rformatted)
  }
  
  # add significance levels if desired
  if (show_significance) {
    # define notions for significance levels; spacing is important.
    stars <- ifelse(is.na(p), "   ", ifelse(p < .001, "***", ifelse(p < .01, "** ", ifelse(p < .05, "*  ", "   "))))
    Rformatted = paste0(Rformatted, stars)
  }
  # build a new matrix that includes the formatted correlations and their significance stars
  Rnew <- matrix(Rformatted, ncol = ncol(x))
  rownames(Rnew) <- colnames(x)
  colnames(Rnew) <- paste(colnames(x), "", sep =" ")
  
  # replace undesired values
  if (use == 'upper') {
    Rnew[lower.tri(Rnew, diag = replace_diagonal)] <- replacement
  } else if (use == 'lower') {
    Rnew[upper.tri(Rnew, diag = replace_diagonal)] <- replacement
  } else if (replace_diagonal) {
    diag(Rnew) <- replacement
  }
  
  return(Rnew)
}

plot_comparison_with_functional_forms <- function(df, x_var, y_var) {
  names <- naming_correction(list(x_var, y_var), c())
  
  plot <- ggplot(data = df, aes_string(x = x_var, y = y_var)) + 
  geom_point() + geom_smooth(method = lm, formula = y~poly(x, 1), col = "red", se = FALSE) + geom_smooth(method = lm, formula = y~poly(x, 2), col = "blue", se = FALSE) + geom_smooth(method = lm, formula = y~poly(x, 3), col = "green", se = FALSE) +
  geom_vline(aes(xintercept = mean(df[[x_var]], na.rm = TRUE)), linetype = 4, color = "red") + 
  labs(title = glue('{names[1]} vs. {names[2]}'), x = names[1], y = names[2])

  testing_model_linear <- lm(as.formula(glue('{y_var} ~ {x_var}')), data = df)
  testing_model_quadratic <- lm(as.formula(glue('{y_var} ~ {x_var} + I({x_var}^2)')), data = df)
  testing_model_cubic <- lm(as.formula(glue('{y_var} ~ {x_var} + I({x_var}^2) + I({x_var}^3)')), data = df)

  degree_counter <- 1
  for (model in list(testing_model_linear, testing_model_quadratic, testing_model_cubic)) {
    mse <- mean(summary(model)$residuals ^ 2)
    print(glue('MSE of Poly w/ Degree {degree_counter}: {mse}'))

    degree_counter <- degree_counter + 1
  }
  return(plot)
}
